# No write reaches a payload that an arm borrows, and no line names the rule
# borrow-match.  A take or a replace of an Option of a payload with a
# destructor, whose result has a destructor too, writes a different sum or
# runs in the arm on_none.  A loop over a sum that only reads it, a write of a
# different Option in the body of a loop, and a write of the sum after the end
# of the loop reach no payload that the loop lends.

compile others src/others.cpp
flag others -freflection
exit others pass

# In the unit referents an arm holds a Ref by a capture by copy: the Ref names a
# different Option, the Ref of the same Option only reads it, and the write
# through the Ref comes after the match.  A borrow whose first type argument is a
# tag and whose statement names its target names a different Option too.
compile referents src/referents.cpp
flag referents -freflection
exit referents pass
