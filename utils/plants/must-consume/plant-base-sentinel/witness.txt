# A body of a base header that is no instance of a template counts only in the
# sentinel unit of the header: the unit whose argument sentinel names the
# header, and the base unit whose main file holds the body.  An instance of a
# template counts in each unit that makes it, and an instance with a type that
# is not linear gives no line.  A parse and an object compile of one unit give
# the same lines.

compile sentinel src/uses.cpp sentinel=include/fixy/Uses.h must-consume=on stage=c
flag sentinel -freflection
exit sentinel pass

compile parse src/uses.cpp sentinel=include/fixy/Uses.h must-consume=on stage=syntax findings=sentinel
