# A base header gets the base kinds in its sentinel unit.  The rule of the
# library names reads it with the admit rows of the language: the row of
# std::move admits it, the row with `in` admits std::size_t in this header,
# and the row of std::bit_cast with `until` and no `in` refuses it.  The
# library object, the C library struct, the C library call and the folded
# name of the static_assert are findings of a base kind.  The raw pointer,
# the C array and the builtin are no finding.  The quarantined unit with no
# sentinel reads no base header, and no unit names Other.h as a sentinel.  In
# the mode error each finding of a base kind is an error.

compile plain src/kinds.cpp stage=c
exit plain pass
judge plain

compile sentinel src/kinds.cpp sentinel=include/fixy/Kinds.h stage=c
exit sentinel pass
judge sentinel
