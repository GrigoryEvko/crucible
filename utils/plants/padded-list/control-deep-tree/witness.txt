# The rule padded-list keeps the answer of each type.  A tree whose levels
# hold the level below them two times then compiles in less than 60 seconds,
# and the longest list of each element type gives one line.  The first
# compile gives the threshold 1, so the plugin gives the line of each list of
# two or more elements.  The members of the union Wide have more value ranges
# than the rule keeps, so the union holds the value bits of its largest
# member, and the line counts more padding bits than the union has.  The
# second compile gives the threshold of utils/registry.txt.  No list has more
# elements than that threshold, so the plugin gives no line of the rule, and
# the module finds no padded list.  The union in base code is also a line of
# the rule proof-routes.
compile sentinel src/tree.cpp sentinel=include/fixy/Tree.h stage=c timeout=60 padded-list-limit=1
exit sentinel pass
