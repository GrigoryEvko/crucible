# No store keeps a borrow of an object that ends with the frame, and no line
# names the rule view-escape: a View of a list of the caller into the object
# of a parameter by reference and into a member, a View into a local, a View
# that a parameter by value holds into a member, a member that a constructor
# builds from a list of its caller, and an assignment operator of namespace
# scope on the object of a parameter by reference.  A first form of the walk
# of the stores read the class of such an operator, which is a namespace, and
# the compile ended with a segmentation fault.  The form of
# fixy/SharedRegion.h gives no line: a View of the run of a region that a
# constructor consumes, because the run lives outside the region.

compile stores src/stores.cpp
flag stores -freflection
exit stores pass

# An object of an empty class goes to a call as a tag, and the result names no
# object, so a constructor that keeps the result keeps no borrow.

compile empty src/empty.cpp
flag empty -freflection
exit empty pass
