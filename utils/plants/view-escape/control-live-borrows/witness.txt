# No use reads a dead object, and no line names the rule view-escape: a use before
# the change, a borrow that the function takes again, a change on a path that
# returns, a move that keeps the storage, a frame that gives the owner back, a call
# by a const reference, a change of a different owner, and a call with an arena
# that keeps its blocks.  The unit blocks has a result that names a store, and not an
# element of the list that the store holds.  The unit typed_owners has owners of the
# base that keep their blocks and that hold no raw block.  In the unit inner_owners an
# owner leaves its wrapper and lives while a borrow of it is in use.  In the unit
# constructs a construct at a pointer moves an owner into an object that lives on.
# In the unit segments a SegmentList of Views keeps a borrow of an element across
# an append and across an append with code of the element.

compile live src/live.cpp
flag live -freflection
exit live pass

compile blocks src/blocks.cpp
flag blocks -freflection
exit blocks pass

compile typed_owners src/typed_owners.cpp
flag typed_owners -freflection
exit typed_owners pass

compile inner_owners src/inner_owners.cpp
flag inner_owners -freflection
exit inner_owners pass

compile constructs src/constructs.cpp
flag constructs -freflection
exit constructs pass

compile segments src/segments.cpp
flag segments -freflection
exit segments pass
