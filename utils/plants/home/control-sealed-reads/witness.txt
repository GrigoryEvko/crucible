# With the rule specialization-home on, a gate that reads a sealed template is
# closed: a class template or a variable template that the unit defines, that
# no row governs and that has no open mark.  The rule specialization-home
# refuses each specialization of such a template outside its home, so no
# unit changes the answer of the gate.  A gate over a variable template, a
# class trait, a member function and a member variable template of a class
# template, and a requires-clause, gives no line, also in a crucible header
# and in a check file.  So does a gate over a class template that the unit
# only declares, when the header holds a specialization of it or when it is a
# template of a base root in a base header, and a gate
# over a sealed template whose definition reads only sealed templates: at a
# second depth, through a template argument, and through a recursion.
compile layer src/layer.cpp mode=error specialization-home=on
flag layer -fplugin-arg-crucible_quarantine-sentinel=include/foundation/gates.h
exit layer pass
compile crucible src/crucible.cpp mode=error specialization-home=on
flag crucible -fplugin-arg-crucible_quarantine-sentinel=include/crucible/gates.h
exit crucible pass
compile check_file test/layer/checks/foundation/gates.cpp mode=error specialization-home=on
exit check_file pass
