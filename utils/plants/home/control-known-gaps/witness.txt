# The rules home, specialization-home and base-namespace give no line for a
# route that changes the answer of a gate of the base: an overload beside the
# own type of a file that argument-dependent lookup finds for a function that
# is not a template.  The rule base-namespace refuses a new overload in a
# namespace of the base and the body of a function template that the base
# only declares (utils/plants/base-namespace).  This control pins the gap.
compile gaps src/gaps.cpp mode=error specialization-home=on base-namespace=on
exit gaps pass
