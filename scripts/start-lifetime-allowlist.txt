# start-lifetime-allowlist — the reviewed uses of std::start_lifetime_as
# and std::start_lifetime_as_array.  scripts/check-start-lifetime.sh reads
# this file.
#
# Format: path:key, or path:key xN.  The key is the name and its template
# argument list, with the white space reduced, as the guard prints it.
# An entry admits exactly one use in that file with that key, or N uses
# with the suffix.  A line that starts with '#' is a comment, and each
# entry has one above it.
#
# A proof type is not an implicit-lifetime type.  A site is safe when its
# element type is a concrete implicit-lifetime type, or a template
# parameter that a constraint holds to one.  The comment of each entry
# names the element type and says which of the two holds.

# The auxiliary buffer of a trace entry: 64-bit scalars and the strong
# IDs OpIndex and SlotId, each a trivially copyable 32-bit wrapper.
include/crucible/BackgroundThread.h:start_lifetime_as_array<int64_t>
include/crucible/BackgroundThread.h:start_lifetime_as_array<OpIndex>
include/crucible/BackgroundThread.h:start_lifetime_as_array<SlotId> x2
bench/bench_phases.cpp:start_lifetime_as_array<int64_t>
bench/bench_phases.cpp:start_lifetime_as_array<OpIndex>
bench/bench_phases.cpp:start_lifetime_as_array<SlotId> x2

# The erasure-code kernels: AVX2 vectors and byte lanes over packet bytes.

include/crucible/cntp/Fec.h:start_lifetime_as<const std::uint8_t> x4

# The seqlock read of a snapshot.  T is a template parameter, and the
# class constraint holds it to a trivially copyable type.  The mandate of
# std::start_lifetime_as also refuses a type that is not implicit-lifetime.
include/crucible/concurrent/AtomicSnapshot.h:start_lifetime_as<T> x2

# A KNOWN OPEN SITE.  slot_type is the template parameter SlotPtr, and
# nothing constrains it, so SwissTableBuffer<P> over a proof type P gives
# slots whose lifetime never started.  Each user in the tree passes a
# pointer type.  The header is in the frozen tree, and no new-tree copy
# exists to mirror, so the constraint lands when the buffer moves to the
# new tree: SlotPtr must be an implicit-lifetime type.
include/crucible/safety/SwissTableBuffer.h:start_lifetime_as_array<slot_type>

# The checked lifetime start, one overload for mutable and one for const
# storage.  T is a template parameter, and each overload requires
# ImplicitLifetimeThroughout<T>: T and each subobject of T at each depth
# is an implicit-lifetime type.  New code uses this function.
include/foundation/Lifetime.h:start_lifetime_as_array<T> x2
